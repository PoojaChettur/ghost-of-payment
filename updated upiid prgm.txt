/// Service class for chat message analysis to detect scam keywords.
class ChatAnalysisService {
  static final Map<String, List<String>> scamKeywords = <String, List<String>>{
    'English': <String>['OTP', 'jackpot', 'free gift'],
    'Hindi': <String>['इनाम', 'फ्री', 'OTP'],
    'Tamil': <String>['வெற்றி', 'இலவசம்', 'OTP'],
    'Telugu': <String>['ఉచితం', 'ఒప్పందం', 'OTP'],
  };

  static String getScamKeywordInMessage(String msg, String lang) {
    final String lowerMsg = msg.toLowerCase();
    final List<String> keywords =
        scamKeywords[lang] ?? <String>[]; // Provide a fallback empty list
    return keywords.firstWhere(
      (String kw) => lowerMsg.contains(kw.toLowerCase()),
      orElse: () => '',
    );
  }
}

/// Service class for detecting scam UPI IDs.
class ScamDetectionService {
  // This list would typically be fetched from a backend or persistent storage.
  static final List<String> scamUpiList = <String>[
    'fraud@ybl',
    'scammer@upi',
    'fake@okaxis',
    'fraud@upi',
  ];

  static bool isScamUPI(String upiId) {
    return scamUpiList.contains(upiId.toLowerCase());
  }

  // A placeholder method for reporting, in a real app this would send data.
  static void reportScamUpi(String upiId) {
    // In a real application, this would send the UPI ID to a backend
    // for review and potential addition to a global scam list.
    // For this simulation, we'll just acknowledge the report.
    debugPrint('Reported UPI ID: $upiId');
  }
}

/// A placeholder screen for the Transaction Simulator.
///
/// This screen provides a basic Scaffold with an AppBar and some descriptive text
/// and an icon, serving as a functional destination for navigation. It now
/// includes input fields for UPI ID and amount, and simulates a transaction
/// with a scam check.
class TransactionSimulatorScreen extends StatefulWidget {
  const TransactionSimulatorScreen({super.key});

  @override
  State<TransactionSimulatorScreen> createState() =>
      _TransactionSimulatorScreenState();
}

class _TransactionSimulatorScreenState
    extends State<TransactionSimulatorScreen> {
  final TextEditingController _upiIdController = TextEditingController();
  final TextEditingController _amountController = TextEditingController();

  @override
  void dispose() {
    _upiIdController.dispose();
    _amountController.dispose();
    super.dispose();
  }

  void _simulateTransaction() {
    final String upiId = _upiIdController.text.trim();
    final String amountText = _amountController.text.trim();

    if (upiId.isEmpty) {
      _showAlertDialog('Input Error', 'Please enter a UPI ID.');
      return;
    }

    if (amountText.isEmpty) {
      _showAlertDialog('Input Error', 'Please enter an amount.');
      return;
    }

    final double? amount = double.tryParse(amountText);
    if (amount == null || amount <= 0) {
      _showAlertDialog(
          'Input Error', 'Please enter a valid amount (e.g., 100.00).');
      return;
    }

    final bool isScam = ScamDetectionService.isScamUPI(upiId);

    _showTransactionResultDialog(upiId, amount, isScam);
  }

  void _showAlertDialog(String title, String message) {
    showDialog<void>(
      context: context,
      builder: (BuildContext dialogContext) {
        return AlertDialog(
          title: Text(title),
          content: Text(message),
          actions: <Widget>[
            TextButton(
              onPressed: () {
                Navigator.of(dialogContext).pop();
              },
              child: const Text('OK'),
            ),
          ],
        );
      },
    );
  }

  void _showTransactionResultDialog(
      String upiId, double amount, bool isScam) {
    showDialog<void>(
      context: context,
      builder: (BuildContext dialogContext) {
        return AlertDialog(
          title: Text(isScam ? 'Scam Alert!' : 'UPI ID Safe'),
          content: Column(
            mainAxisSize: MainAxisSize.min,
            crossAxisAlignment: CrossAxisAlignment.start,
            children: <Widget>[
              Text('UPI ID: $upiId'),
              Text('Amount: ₹${amount.toStringAsFixed(2)}'),
              const SizedBox(height: 10),
              Text(
                isScam
                    ? '⚠️ This UPI ID is identified as a potential scam. Proceed with extreme caution!'
                    : '✅ This UPI ID appears safe. You can proceed with the transaction.',
                style: TextStyle(
                  color: isScam ? Colors.red : Colors.green,
                  fontWeight: FontWeight.bold,
                ),
              ),
            ],
          ),
          actions: <Widget>[
            TextButton(
              onPressed: () {
                Navigator.of(dialogContext).pop(); // Dismiss the dialog
              },
              child: const Text('Cancel'),
            ),
            ElevatedButton(
              onPressed: () {
                Navigator.of(dialogContext).pop(); // Dismiss the dialog
                // In a real app, this would trigger the actual transaction
                _showAlertDialog(
                    'Transaction Simulated',
                    'The transaction for ₹${amount.toStringAsFixed(2)} to $upiId has been simulated.');
              },
              child: const Text('Proceed'),
            ),
          ],
        );
      },
    );
  }

  @override
  Widget build(BuildContext context) {

      appBar: AppBar(

        backgroundColor: Theme.of(context).appBarTheme.backgroundColor,

        padding: const EdgeInsets.all(20.0),
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: <Widget>[
            Icon(
              Icons.account_balance_wallet,
              size: 80,
              color: Theme.of(context).colorScheme.primary,

            const SizedBox(height: 20),
            Text(
              'Simulate transactions safely.',
              style: TextStyle(
                fontSize: 24,
                fontWeight: FontWeight.bold,
                color: Theme.of(context).colorScheme.onSurface,

              textAlign: TextAlign.center,

              style: TextStyle(
                fontSize: 16,

              ),
              textAlign: TextAlign.center,

class _ChatAnalyzerScreenState extends State<ChatAnalyzerScreen> {
  final TextEditingController _chatController = TextEditingController();
  String selectedLanguage = 'English';
  String analysisResult = '';

  @override
  void dispose() {
    _chatController.dispose(); // Dispose the controller
    super.dispose();
  }

  void analyzeChat() {
    final String message = _chatController.text.trim();
    final String keyword =
        ChatAnalysisService.getScamKeywordInMessage(message, selectedLanguage);

    setState(() {
      analysisResult = keyword.isNotEmpty
          ? '🔍 Scam keyword found: $keyword'
          : '✅ Message appears safe.';
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Chat Analyzer'),
        backgroundColor: Theme.of(context).appBarTheme.backgroundColor,
        foregroundColor: Theme.of(context).appBarTheme.foregroundColor,
      ),
      body: Padding(
        padding: const EdgeInsets.all(16),
        child: Column(
          children: <Widget>[
            DropdownButtonFormField<String>(
              value: selectedLanguage,
              items: <String>['English', 'Hindi', 'Tamil', 'Telugu']
                  .map<DropdownMenuItem<String>>((String lang) {
                return DropdownMenuItem<String>(value: lang, child: Text(lang));
              }).toList(),
              onChanged: (String? val) {
                if (val != null) {
                  setState(() => selectedLanguage = val);
                }
              },
              decoration: const InputDecoration(
                labelText: 'Select Language',
                border: OutlineInputBorder(),
              ),
            ),
            const SizedBox(height: 12),
            TextField(
              controller: _chatController,
              decoration: const InputDecoration(
                hintText: 'Type your message here…',
                border: OutlineInputBorder(),
                prefixIcon: Icon(Icons.message),
              ),
              maxLines: null, // Allows multiline input
              minLines: 5, // Ensures a reasonable initial height
            ),
            const SizedBox(height: 16),
            ElevatedButton(
              onPressed: analyzeChat,
              child: const Text('Analyze Message'),
            ),
            if (analysisResult.isNotEmpty)
              Padding(
                padding: const EdgeInsets.only(top: 12),
                child:
                    Text(analysisResult, style: const TextStyle(fontSize: 16)),
              ),
          ],
        ),
      ),
    );
  }
}

/// The initial splash screen displayed when the app starts.
///
/// It shows the app's branding and automatically navigates to the HomeScreen
/// after a short delay. Navigation is handled safely using `addPostFrameCallback`.
class SplashScreen extends StatefulWidget {
  const SplashScreen({super.key});

  @override
  State<SplashScreen> createState() => _SplashScreenState();
}

class _SplashScreenState extends State<SplashScreen>
    with SingleTickerProviderStateMixin {
  late AnimationController _controller;
  late Animation<double> _iconFadeAnimation;
  late Animation<double> _textFadeAnimation;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(
      vsync: this,
      duration: const Duration(seconds: 2), // Total animation duration
    );

    // Icon fades out in the first half of the animation
    _iconFadeAnimation = Tween<double>(begin: 1.0, end: 0.0).animate(
      CurvedAnimation(
        parent: _controller,
        curve: const Interval(0.0, 0.5, curve: Curves.easeOut),
      ),
    );

    // Text fades in in the second half of the animation
    _textFadeAnimation = Tween<double>(begin: 0.0, end: 1.0).animate(
      CurvedAnimation(
        parent: _controller,
        curve: const Interval(0.5, 1.0, curve: Curves.easeIn),
      ),
    );

    _controller.forward(); // Start the animation

    _controller.addStatusListener((AnimationStatus status) {
      if (status == AnimationStatus.completed) {
        // Navigate to HomeScreen after animation completes
        if (Navigator.of(context).mounted) {
          Navigator.pushReplacement<void, void>(
            context,
            MaterialPageRoute<void>(builder: (_) => const HomeScreen()),
          );
        }
      }
    });
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Center(
        child: Stack(
          alignment: Alignment.center,
          children: <Widget>[
            // Shield Icon - Fades out
            FadeTransition(
              opacity: _iconFadeAnimation,
              child: Icon(
                Icons.shield,
                size: 100,
                color: Theme.of(context).colorScheme.primary,
              ),
            ),
            // Welcome Text - Fades in
            FadeTransition(
              opacity: _textFadeAnimation,
              child: Text(
                'Welcome to Scam Shield AI',
                style: TextStyle(
                  fontSize: 28,
                  fontWeight: FontWeight.bold,
                  color: Theme.of(context).colorScheme.primary,
                ),
                textAlign: TextAlign.center,
              ),
            ),
          ],
        ),
      ),
    );
  }
}

/// Screen for scanning QR codes and checking them for scam indicators.
class QrScannerScreen extends StatefulWidget {
  const QrScannerScreen({super.key});

  @override
  State<QrScannerScreen> createState() => _QrScannerScreenState();
}

class _QrScannerScreenState extends State<QrScannerScreen> {
  String? _simulatedQrContent; // Holds the content that was "scanned"
  String _scanResult = '';
  bool _isScanning = false; // To show loading state during simulation

  // DATA_MODEL for simulated QR codes
  static const List<String> _sampleQrContents = <String>[
    'upi://pay?pa=safeuser@bank',
    'upi://pay?pa=scammer@upi', // Known scam
    'upi://pay?pa=merchant@okicici',
    'upi://pay?pa=fraud@ybl', // Known scam
    'https://example.com/safe-link',
    'https://malicious.example.com/phishing', // Placeholder for a "bad" link
    'Just some random text content',
  ];

  @override
  void dispose() {
    super.dispose();
  }

  // New method to simulate the scan
  Future<void> _simulateScan() async {
    setState(() {
      _isScanning = true;
      _scanResult = ''; // Clear previous result
      _simulatedQrContent = null; // Clear previous content display
    });

    // Simulate network/camera delay
    await Future<void>.delayed(const Duration(seconds: 2));

    if (!mounted) {
      return;
    }

    // Pick a random QR content
    final String pickedContent =
        _sampleQrContents[DateTime.now().microsecondsSinceEpoch % _sampleQrContents.length];

    setState(() {
      _simulatedQrContent = pickedContent;
      _isScanning = false;
    });

    // Now analyze the picked content
    _analyzeQrContent(pickedContent);
  }

  // Analyzes the provided QR content for scam indicators
  void _analyzeQrContent(String qrContent) {
    if (qrContent.isEmpty) {
      setState(() {
        _scanResult = 'No QR code content detected.';
      });
      return;
    }

    // Check if it's a UPI ID first
    if (qrContent.contains('@')) {
      // Extract UPI ID if it's a full UPI URL like upi://pay?pa=...
      String upiId = qrContent;
      final Uri? uri = Uri.tryParse(qrContent);
      if (uri != null && uri.scheme == 'upi' && uri.queryParameters.containsKey('pa')) {
        upiId = uri.queryParameters['pa']!;
      }

      final bool isScamUpi = ScamDetectionService.isScamUPI(upiId);
      setState(() {
        if (isScamUpi) {
          _scanResult =
              '⚠️ Scam Alert! This UPI ID ($upiId) is identified as a potential scam. DO NOT PROCEED!';
        } else {
          _scanResult =

      }
